Step 3 -- Build PM-specific commands. Use `ctx.packageManager`:

| PM | Install command | Run prefix | Cache key | Needs corepack | Action |
|----|----------------|------------|-----------|----------------|--------|
| npm | `npm ci` | `npm run` | `npm` | no | none |
| pnpm | `pnpm install --frozen-lockfile` | `pnpm run` | `pnpm` | yes | `pnpm/action-setup@v4` with `version: 10` |
| yarn | `yarn install --frozen-lockfile` | `yarn run` | `yarn` | yes | none (corepack handles it) |
| bun | `bun install --frozen-lockfile` | `bun run` | none | no | `oven-sh/setup-bun@v2` |

Step 4 -- Generate YAML content as a string template. Use Claude's discretion on trigger: both PR and push to main (per research recommendation).

Template structure:
```yaml
name: CI

on:
  pull_request:
    branches: [main]
  push:
    branches: [main]

jobs:
  ci:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        node-version: [20, 22]
    steps:
      - uses: actions/checkout@v4

      # PM-specific setup step (only for pnpm or bun)
      # pnpm: uses pnpm/action-setup@v4 with version: 10
      # bun: uses oven-sh/setup-bun@v2

      - uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: '<pm>'  # npm, pnpm, or yarn (omit for bun)

      # Corepack enable (pnpm/yarn only)
      - run: corepack enable

      - run: <install-command>

      # Conditional steps (only included if detected)
      - run: <run-prefix> lint          # if hasLint
      - run: <run-prefix> typecheck     # if hasTypecheck
      - run: <run-prefix> test -- --run # if hasTest (--run for vitest)
      - run: <run-prefix> build         # if hasBuild
```

For bun: omit the `actions/setup-node` step entirely, replace with `oven-sh/setup-bun@v2`. No `cache` or `corepack` needed. Omit `node-version` matrix (bun has its own version). Use `bun run` prefix and `bun install --frozen-lockfile`.

For the test step: if `vitest` is in deps, append `-- --run` to ensure vitest runs in non-watch mode. If `jest` or unknown test runner, just use `<run-prefix> test`.

**2. Extend `packages/core/src/enhancements/summary.ts`:**

Keep the existing `showEnhancementSummary()` function unchanged. Add a new `showPerEnhancementSummary()` function for per-module detail cards (per locked decision).

```typescript
import type { InstallResult } from './types.js'

/** Next steps hint per enhancement module */
export interface EnhancementNextSteps {
  moduleId: string
  moduleName: string
  result: InstallResult
  nextSteps: string[]
}

/**
 * Display a detailed per-enhancement summary card.
 *
 * Shows: module name, files created/modified, packages installed, next steps.
 * Called once per successfully installed module.
 */
export function showPerEnhancementSummary(info: EnhancementNextSteps): void {
  tinkeriseBlankLine()
  tinkeriseLog(pc.bold(info.moduleName))

  if (info.result.filesModified.length > 0) {
    tinkeriseLog(`  ${pc.dim('Files:')} ${info.result.filesModified.join(', ')}`)
  }

  if (info.result.packagesAdded.length > 0) {
    tinkeriseLog(`  ${pc.dim('Packages:')} ${info.result.packagesAdded.join(', ')}`)
  }

  if (info.result.warnings.length > 0) {
    for (const w of info.result.warnings) {
      tinkeriseLog(`  ${pc.yellow('!')} ${w}`)
    }
  }

  if (info.nextSteps.length > 0) {
    tinkeriseLog(`  ${pc.dim('Next:')}`)
    for (const step of info.nextSteps) {
      tinkeriseLog(`    ${pc.cyan('>')} ${step}`)
    }
  }
}
```
