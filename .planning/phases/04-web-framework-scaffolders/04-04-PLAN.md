```typescript
import { describe, it, expect, beforeAll, afterAll } from 'vitest'
import { mkdtemp, rm, access } from 'node:fs/promises'
import { tmpdir } from 'node:os'
import { join } from 'node:path'
import { execa } from 'execa'

const E2E_ENABLED = process.env.TINKERISE_E2E === 'true'

// Path to the built CLI binary
const CLI_PATH = join(__dirname, '../../dist/index.js')

// Long timeout for E2E tests (scaffolding + optional install can take time)
const E2E_TIMEOUT = 180_000  // 3 minutes
```

For each scaffolder, create a `describe.skipIf(!E2E_ENABLED)` block:

3. **Astro** — `tinkerise web astro test-astro --no-install --no-git -- --yes` (pass --yes to skip Astro's own prompts)

   - Verify: `test-turbo` directory created, `turbo.json` exists

Each test follows this pattern:
```typescript
describe.skipIf(!E2E_ENABLED)('E2E: scaffold [name]', () => {
  let tmpDir: string

  beforeAll(async () => {
    tmpDir = await mkdtemp(join(tmpdir(), 'tinkerise-e2e-'))
  })

  afterAll(async () => {
    await rm(tmpDir, { recursive: true, force: true })
  })

  it('scaffolds a [framework] project', async () => {
    const result = await execa('node', [
      CLI_PATH, ...args
    ], {
      cwd: tmpDir,
      timeout: E2E_TIMEOUT,
      env: { ...process.env, NODE_ENV: 'test' },
    })

    expect(result.exitCode).toBe(0)

    // Verify project directory exists
    const projectDir = join(tmpDir, projectName)
    await expect(access(projectDir)).resolves.toBeUndefined()

    // Verify key files exist
    await expect(access(join(projectDir, 'package.json'))).resolves.toBeUndefined()
  }, E2E_TIMEOUT)
})
```

Important implementation notes:
- Use `stdio: 'pipe'` instead of `inherit` for E2E tests so output doesn't pollute test runner
- Use `reject: false` from execa so we can check exit code
- Wait — actually the E2E test calls the CLI binary which internally uses `stdio: inherit` for the subprocess. When the CLI runs under execa with `stdio: 'pipe'`, the subprocess's inherit will try to inherit the pipe, which should work. Alternatively, just check exit code only.
- Each describe block gets its own tmpDir for isolation
- Use `--no-install` and `--no-git` where supported to speed up tests
- For upstream tools that need their own non-interactive flags (--yes, --CI, -y), pass via `--` passthrough
