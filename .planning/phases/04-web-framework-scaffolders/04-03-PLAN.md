Implement the `tinkerise list` command per locked decisions:
- Default `tinkerise list`: shows minimal info — scaffolder names grouped by category
- `tinkerise list web`: filters to a category, shows more detail (description, package name)
- Show prerequisite status next to each scaffolder (checkmark or X)

```typescript
import pc from 'picocolors'
import type { ScaffolderCategory } from '@tinkerise/shared'
import { getAllScaffolders, getScaffoldersByCategory, checkPrerequisite } from '@tinkerise/core'
import type { ScaffolderEntry } from '@tinkerise/shared'
import { getScaffolderMetadata } from '@tinkerise/core'

const CATEGORY_ORDER: ScaffolderCategory[] = ['web', 'backend', 'mobile']
const CATEGORY_LABELS: Record<string, string> = {
  web: 'Web',
  backend: 'Backend',
  mobile: 'Mobile',
}

/**
 * Check if all prerequisites for a scaffolder are met.
 * Uses the first prerequisite only for quick status display.
 */
async function checkPrereqStatus(entry: ScaffolderEntry): Promise<boolean> {
  if (entry.prerequisites.length === 0) return true
  try {
    for (const prereq of entry.prerequisites) {
      const result = await checkPrerequisite(prereq)
      if (!result.satisfied) return false
    }
    return true
  } catch {
    return false
  }
}

/**
 * List all available scaffolders, optionally filtered by category.
 */
export async function listScaffolders(filterCategory?: string): Promise<void> {
  // Validate category if provided
  if (filterCategory && !CATEGORY_ORDER.includes(filterCategory as ScaffolderCategory)) {
    console.error(pc.red(`Unknown category: '${filterCategory}'. Valid: ${CATEGORY_ORDER.join(', ')}`))
    process.exit(1)
  }

  const scaffolders = filterCategory
    ? getScaffoldersByCategory(filterCategory as ScaffolderCategory)
    : getAllScaffolders()

  if (scaffolders.length === 0) {
    console.log(pc.dim('No scaffolders available.'))
    return
  }

  // Group by category
  const grouped = new Map<string, ScaffolderEntry[]>()
  for (const s of scaffolders) {
    const list = grouped.get(s.category) ?? []
    list.push(s)
    grouped.set(s.category, list)
  }

  for (const cat of CATEGORY_ORDER) {
    const items = grouped.get(cat)
    if (!items || items.length === 0) continue

    const label = CATEGORY_LABELS[cat] ?? cat
    console.log(`\n${pc.bold(label)}`)

    for (const item of items) {
      const prereqOk = await checkPrereqStatus(item)
      const icon = prereqOk ? pc.green('✓') : pc.red('✗')
      const metadata = getScaffolderMetadata(item.name)
      const displayName = metadata?.displayName ?? item.name

      if (filterCategory) {
        // Detailed view: name, description, package
        console.log(`  ${icon} ${pc.bold(displayName)}`)
        if (metadata?.description) {
          console.log(`    ${pc.dim(metadata.description)}`)
        }
        console.log(`    ${pc.dim(`Package: ${item.packageName}`)}`)

        // Show supported unified flags
        const supportedFlags = item.flags
          .filter(f => f.native !== '')  // Exclude silent/no-op flags
          .map(f => `--${f.unified}`)
        if (supportedFlags.length > 0) {
          console.log(`    ${pc.dim(`Flags: ${supportedFlags.join(', ')}`)}`)
        }
      } else {
        // Minimal view: just name
        console.log(`  ${icon} ${displayName}`)
      }
    }
  }
  console.log()  // trailing newline
}
```

**Per locked decision:** `tinkerise web next --help` shows only tinkerise's unified flags — native flags work but aren't advertised in help output.

Commander.js by default shows all registered options in `--help`. We need to customize the help output for per-scaffolder subcommands so that only the scaffolder's supported unified flags appear, and native/passthrough flags are not shown.

**Create `packages/cli/src/commands/help.ts`:**

Implement a `scaffolderHelp()` function that generates custom help text for a given scaffolder:

```typescript
import { getScaffolder } from '@tinkerise/core'
import { getScaffolderMetadata } from '@tinkerise/core'

/**
 * Build custom help text for a scaffolder showing only unified flags.
 * Per locked decision: native flags work but aren't advertised.
 */
export function buildScaffolderHelpText(scaffolderName: string): string {
  const entry = getScaffolder(scaffolderName)
  if (!entry) return ''

  const metadata = getScaffolderMetadata(scaffolderName)
  const displayName = metadata?.displayName ?? scaffolderName

  const lines: string[] = []
  lines.push(`Usage: tinkerise web ${scaffolderName} <name> [options]`)
  lines.push('')
  lines.push(`Scaffold a new ${displayName} project`)
  lines.push('')
  lines.push('Options:')

  // Show only unified flags that are supported by this scaffolder
  // Skip silent/no-op flags (empty native) as they are auto-handled
  for (const flag of entry.flags) {
    if (flag.native === '') {
      // Silent flag — still show it as supported (user can pass it)
      lines.push(`  --${flag.unified}${' '.repeat(Math.max(1, 20 - flag.unified.length))}(always enabled, accepted silently)`)
    } else {
      lines.push(`  --${flag.unified}`)
    }
  }

  // Always show universal flags
  lines.push(`  --no-install${' '.repeat(8)}Skip dependency installation`)
  lines.push(`  --no-git${' '.repeat(12)}Skip git initialization`)
  lines.push(`  --package-manager <pm>  Package manager (npm, pnpm, yarn, bun)`)
  lines.push(`  --  <args>              Pass additional args to ${displayName}`)
  lines.push(`  -h, --help              Display this help`)
  lines.push('')
  lines.push('Native upstream flags are also accepted but not shown here.')

  return lines.join('\n')
}
```

Wait — the above approach duplicates the universal flags. Better approach: read the scaffolder's actual `flags` array and show ONLY those. Flags not in the array are unsupported (will error), so they shouldn't appear in help. The `--` passthrough and `--help` are always available.

Revised approach — show exactly what the scaffolder supports:

```typescript
export function buildScaffolderHelpText(scaffolderName: string): string {
  const entry = getScaffolder(scaffolderName)
  if (!entry) return ''

  const metadata = getScaffolderMetadata(scaffolderName)
  const displayName = metadata?.displayName ?? scaffolderName

  const lines: string[] = []
  lines.push(`Usage: tinkerise web ${scaffolderName} <name> [options]`)
  lines.push('')
  lines.push(`Scaffold a new ${displayName} project`)
  if (metadata?.description) lines.push(metadata.description)
  lines.push('')
  lines.push('Supported flags:')

  for (const flag of entry.flags) {
    const flagStr = `  --${flag.unified}`
    if (flag.native === '') {
      lines.push(`${flagStr}${' '.repeat(Math.max(1, 24 - flagStr.length))}(always enabled)`)
    } else {
      lines.push(flagStr)
    }
  }

  if (entry.flags.length === 0) {
    lines.push('  (no scaffolder-specific flags)')
  }

  lines.push('')
  lines.push('Passthrough:')
  lines.push(`  -- <args>               Pass additional args directly to ${displayName}`)
  lines.push('')
  lines.push(`Native ${displayName} flags also accepted via passthrough.`)

  return lines.join('\n')
}
```

**Update `packages/cli/src/index.ts`:**

For each scaffolder subcommand registered under `tinkerise web <framework>`, override Commander's default `--help` behavior. When the user runs `tinkerise web next --help`, intercept and show the custom unified-only help text instead of Commander's auto-generated help.

Commander supports a `.helpInformation()` override or a `.addHelpText('beforeAll', ...)` hook. The cleanest approach: on each framework subcommand, listen to the `--help` option and print custom text:

```typescript
import { buildScaffolderHelpText } from './commands/help.js'

// When registering each web framework subcommand:
frameworkCmd.option('-h, --help', 'Show help')
frameworkCmd.on('option:help', () => {
  console.log(buildScaffolderHelpText(frameworkName))
  process.exit(0)
})
```

Or alternatively, use Commander's `configureHelp()` to override the format. The simplest approach that suppresses Commander's default flags display:

```typescript
// For each web framework subcommand
frameworkCmd.helpOption(false)  // Disable Commander's default --help
frameworkCmd.option('-h, --help', 'Display help')
frameworkCmd.hook('preAction', (thisCmd) => {
  if (thisCmd.opts().help) {
    console.log(buildScaffolderHelpText(frameworkName))
    process.exit(0)
  }
})
```

This ensures that `tinkerise web next --help` shows only the unified flags from the registry entry, not Commander's auto-generated list of all registered options.
